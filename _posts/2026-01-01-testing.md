-- Recursive Fibonacci with memo-ish behavior
fib :: Int -> Int
fib n
  | n < 0     = error "Negative numbers anger the recursion gods"
  | n == 0    = 0
  | n == 1    = 1
  | otherwise = fib (n - 1) + fib (n - 2)

use std::collections::HashMap;
use std::thread;
use std::time::Duration;

#[derive(Debug, Clone)]
enum Entity {
    Warrior { name: String, strength: u32 },
    Mage { name: String, mana: u32 },
    Rogue { name: String, agility: u32 },
}

impl Entity {
    fn power(&self) -> u32 {
        match self {
            Entity::Warrior { strength, .. } => strength * 2,
            Entity::Mage { mana, .. } => mana * 3,
            Entity::Rogue { agility, .. } => agility * 2 + 10,
        }
    }

    fn name(&self) -> &str {
        match self {
            Entity::Warrior { name, .. } => name,
            Entity::Mage { name, .. } => name,
            Entity::Rogue { name, .. } => name,
        }
    }
}

fn simulate_battle(entities: &[Entity]) -> HashMap<String, u32> {
    let mut results = HashMap::new();

    for entity in entities {
        println!("{} enters the arena...", entity.name());
        thread::sleep(Duration::from_millis(200));

        let score = entity.power() + rand::random::<u8>() as u32;
        results.insert(entity.name().to_string(), score);
    }

    results
}

fn main() {
    let party = vec![
        Entity::Warrior {
            name: "Thorin".into(),
            strength: 40,
        },
        Entity::Mage {
            name: "Elora".into(),
            mana: 30,
        },
        Entity::Rogue {
            name: "Shade".into(),
            agility: 25,
        },
    ];

    let results = simulate_battle(&party);

    println!("\nFinal scores:");
    for (name, score) in results {
        println!("{} => {}", name, score);
    }
}
